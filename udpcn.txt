------------------------------------------------- -- Create EMPLOYEE table with Constraints ------------------------------------------------- 
CREATE TABLE EMPLOYEE ( 
Emp_ID NUMBER PRIMARY KEY, 
Name VARCHAR2(20) NOT NULL, 
Email VARCHAR2(30) UNIQUE, 
Salary NUMBER DEFAULT 5000, 
Age NUMBER CHECK (Age >= 18) 
);
