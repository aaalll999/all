------------------------------------------------- -- Create DEPARTMENT table ------------------------------------------------- 
CREATE TABLE DEPARTMENT ( 
Dept_ID int PRIMARY KEY, 
Dept_Name VARCHAR2(20) 
);
------------------------------------------------- -- Create STUDENT table ------------------------------------------------- 
CREATE TABLE STUDENT ( 
RollNo int PRIMARY KEY, 
Name VARCHAR2(20), 
City VARCHAR2(20), 
Dept_ID int, 
 
FOREIGN KEY (Dept_ID) 
REFERENCES DEPARTMENT(Dept_ID) 
);
